### TestNG

@Test
public void Question11() {

    DBUtility.executeQuery("""
            SELECT *
            FROM salaries
            WHERE emp_no = 10102
            ORDER BY from_date;
            """);

    List<HashMap<String, String>> data = DBUtility.getListData();

    Assert.assertTrue(data.size() > 1);

    for (int i = 1; i < data.size(); i++) {

        int previousSalary = Integer.parseInt(data.get(i - 1).get("salary"));
        int currentSalary = Integer.parseInt(data.get(i).get("salary"));

        Assert.assertTrue(currentSalary >= previousSalary);

    }

}
